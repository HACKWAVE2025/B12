document.addEventListener('DOMContentLoaded', function() { const loginForm = document.getElementById('login-form'); const usernameInput = document.getElementById('username'); const passwordInput = document.getElementById('password'); const loginError = document.getElementById('login-error'); // Check if already logged in, redirect to index.html if (localStorage.getItem('loggedIn') === 'true') { window.location.href = 'index.html'; } loginForm.addEventListener('submit', function(event) { event.preventDefault(); // Prevent default form submission const username = usernameInput.value; const password = passwordInput.value; // Simulate authentication if (username === 'user' && password === 'password') { localStorage.setItem('loggedIn', 'true'); // Set logged in state window.location.href = 'index.html'; // Redirect to main app } else { loginError.textContent = 'Invalid username or password.'; loginError.style.display = 'block'; } }); });
